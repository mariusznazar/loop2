async (page) => {
  const m = () => page.evaluate(() => {
    const a = document.activeElement;
    return { y: Math.round(scrollY), bar: Math.round(document.querySelector("header").getBoundingClientRect().top), h: document.documentElement.scrollHeight, hash: location.hash, focus: a ? a.tagName.toLowerCase() + "." + [...a.classList].join(".") : null };
  });
  const out = [];
  await page.evaluate(() => window.scrollTo(0, 9999));
  await page.waitForTimeout(200);
  out.push(["run scrolled", await m()]);
  await page.evaluate(() => [...document.querySelectorAll("header a")].find((x) => x.textContent.trim() === "Programy").click());
  await page.waitForTimeout(400);
  out.push(["programy via bar", await m()]);
  await page.evaluate(() => window.scrollTo(0, 9999));
  await page.waitForTimeout(200);
  out.push(["programy scrolled", await m()]);
  await page.evaluate(() => [...document.querySelectorAll("header a")].find((x) => /Wróć do przejścia/.test(x.textContent)).click());
  await page.waitForTimeout(400);
  out.push(["back to run via bar", await m()]);
  await page.goBack();
  await page.waitForTimeout(400);
  out.push(["history back", await m()]);
  await page.goForward();
  await page.waitForTimeout(400);
  out.push(["history forward to run", await m()]);
  return JSON.stringify(out);
}
