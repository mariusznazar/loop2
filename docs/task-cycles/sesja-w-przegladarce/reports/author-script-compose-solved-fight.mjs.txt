// Working copy of the cycle. Composes a map run of "Las — głębiej" that the map's condition solves in
// the minutes of a wolf's attack. The simulation's own functions pass every game minute; between
// minutes the body is set back to full by hand (a field edit the report names), so that the mage
// lives and the streak grows, until two attacks are beaten and a third comes after minute 545. The
// document is written two minutes before that third attack, its streak set so that it reaches 540
// three minutes after the attack. Usage: node compose-solved-fight.mjs SAVE.json OUT.json [seedMs]
import { pathToFileURL } from "node:url";
import { join } from "node:path";
import { readFileSync, writeFileSync } from "node:fs";

const sim = join(process.cwd(), "dist", "browser", "sim");
const load = (name) => import(pathToFileURL(join(sim, name)).href);
const { send, advanceMinutes, withSaveTime } = await load("game.js");
const { savedGame, loadSavedGame } = await load("saved-game.js");

const [, , input, output, seedText = "1000"] = process.argv;
const doc = JSON.parse(readFileSync(input, "utf8"));
const loaded = loadSavedGame({ version: doc.version, game: doc.game });
if (!loaded.ok) throw new Error("load refused");
const r = (id, condition, action) => ({ id, condition, action });
const program = {
  groups: [
    {
      id: 1,
      name: "Program",
      condition: { kind: "always" },
      rules: [
        r(1, { kind: "always" }, { kind: "kick" }),
        r(2, { kind: "itemBelow", item: "stick", value: 8 }, { kind: "gather", resource: "stick" }),
        r(3, { kind: "always" }, { kind: "explore" }),
      ],
    },
  ],
  nextGroupId: 2,
  nextRuleId: 4,
  tripPlan: { take: [], bring: [] },
};
const base = loaded.state;
const id = base.nextProgramId;
let state = send(
  { ...base, programs: [...base.programs, { id, name: "Kopie", program }], nextProgramId: id + 1 },
  { mapId: "forestDeep", characterId: 1, programId: id },
  Number(seedText),
).state;
const full = (s) => ({
  ...s,
  run: { ...s.run, streak: Math.min(s.run.streak, 300), body: { ...s.run.body, health: 100, satiety: 90, temperature: 37 } },
});
const history = [state];
let attack = null;
for (let minute = 0; minute < 5000; minute += 1) {
  const result = advanceMinutes(state, 1);
  state = result.state;
  if (state.run.death || state.run.completion) throw new Error(`ended at ${state.run.minute}`);
  const attacked = result.events.some((event) => event.kind === "attacked");
  if (attacked && state.run.defeatedAttacks === 2 && state.run.minute > 545) {
    attack = state.run.minute;
    break;
  }
  if (!state.run.activeCombat) state = full(state);
  history.push(state);
}
if (attack === null) throw new Error("no third attack");
let pick = history[history.length - 2];
const streak = 535;
pick = {
  ...pick,
  run: {
    ...pick.run,
    streak,
    longestStreak: Math.max(pick.run.longestStreak, streak),
  },
};
const saved = savedGame(withSaveTime(pick, Date.now()));
const check = loadSavedGame(JSON.parse(JSON.stringify(saved)));
process.stderr.write(
  `attack at ${attack}, saved at ${pick.run.minute}, beaten ${pick.run.defeatedAttacks}, load ${check.ok}\n`,
);
writeFileSync(output, JSON.stringify(saved));
