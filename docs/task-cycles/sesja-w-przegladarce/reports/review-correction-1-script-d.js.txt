async (page) => {
  const m = () => page.evaluate(() => {
    const a = document.activeElement;
    return { y: Math.round(scrollY), bar: Math.round(document.querySelector("header").getBoundingClientRect().top), h: document.documentElement.scrollHeight, hash: location.hash, focus: a ? a.tagName.toLowerCase() + "." + [...a.classList].join(".") : null };
  });
  const out = [];
  await page.getByRole("link", { name: "Mapy", exact: true }).click();
  await page.getByRole("link", { name: /Skraj lasu/ }).first().click();
  await page.waitForTimeout(300);
  await page.evaluate(() => window.scrollTo(0, 0));
  out.push(["mapy top", await m()]);
  const b = page.getByRole("button", { name: "Wejdź: Ala, mapa Skraj lasu" });
  out.push(["Ala Wejdz box", await b.boundingBox()]);
  await b.focus();
  await page.keyboard.press("Enter");
  await page.waitForTimeout(500);
  out.push(["after Wejdz Ala by keyboard", await m()]);
  await page.keyboard.press("Tab");
  out.push(["after Tab", await m()]);
  return JSON.stringify(out);
}
