"""Working copy: compares the rules of the game's stylesheets before and after the split of PR #145.

Reads each commit's stylesheets in the order src/browser/index.html links them (by git show), drops
comments, splits them into rules with their @media or @keyframes context, and compares the multisets
of (context, selector, declarations). Writes the result to the file named as its last argument.
"""

import re
import subprocess
import sys
from collections import Counter


def show(commit, path):
    return subprocess.run(
        ["git", "show", f"{commit}:{path}"], capture_output=True, text=True, encoding="utf-8", check=True
    ).stdout


def sheets(commit):
    html = show(commit, "src/browser/index.html")
    names = re.findall(r'<link[^>]+href="/([^"]+\.css)"', html)
    return [(n, show(commit, f"src/browser/{n}")) for n in names]


def rules(text):
    text = re.sub(r"/\*.*?\*/", "", text, flags=re.S)
    out = []

    def walk(body, context):
        i = 0
        while i < len(body):
            j = body.find("{", i)
            if j < 0:
                break
            head = " ".join(body[i:j].split())
            depth, k = 1, j + 1
            while depth:
                depth += {"{": 1, "}": -1}.get(body[k], 0)
                k += 1
            inner = body[j + 1 : k - 1]
            if head.startswith("@media") or head.startswith("@keyframes") or head.startswith("@supports"):
                walk(inner, context + (head,))
            else:
                decls = tuple(sorted(" ".join(d.split()) for d in inner.split(";") if d.strip()))
                for selector in head.split(","):
                    out.append((context, " ".join(selector.split()), decls))
            i = k

    walk(text, ())
    return out


def collect(commit):
    every = []
    for name, text in sheets(commit):
        every.extend(rules(text))
    return every


before, after = collect(sys.argv[1]), collect(sys.argv[2])
first_a = {}
for idx, r in enumerate(after):
    first_a.setdefault(r, idx)
props = lambda r: {d.split(":")[0].strip() for d in r[2]}
last = lambda sel: set(re.findall(r"\.([\w-]+)", re.split(r"[\s>+~]+", sel.strip())[-1]))
uniq = list(dict.fromkeys(before))
allpairs, shared_cls = [], 0
for i, r1 in enumerate(uniq):
    for r2 in uniq[i + 1 :]:
        if first_a[r2] < first_a[r1] and r1[0] == r2[0]:
            shared = props(r1) & props(r2)
            if shared:
                allpairs.append((r1[1], r2[1], sorted(shared)))
                if last(r1[1]) & last(r2[1]):
                    shared_cls += 1
with open(sys.argv[3], "w", encoding="utf-8", newline="\n") as h:
    h.write(f"reversed pairs setting a same property, same context: {len(allpairs)}; with shared class in last compound: {shared_cls}\n")
    for p in allpairs:
        h.write(f"  {p}\n")
print(len(allpairs), shared_cls)
