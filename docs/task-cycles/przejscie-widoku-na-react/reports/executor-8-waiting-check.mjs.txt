import { assert, inAct, press, withPage } from "../../tests/view/page-document.mjs";
import { test } from "node:test";

function held() {
  const calls = [];
  const call = (name) => () => new Promise((resolve) => calls.push({ name, answer: resolve }));
  return {
    calls,
    client: { currentAccount: call("currentAccount"), loadSavedGame: call("loadSavedGame"), confirmEmail: call("confirmEmail"), reportBrowserFailure: () => Promise.resolve({ ok: true, value: null }) },
    async answer(name, result) {
      const i = calls.findIndex((c) => c.name === name);
      const [c] = calls.splice(i, 1);
      c.answer(result);
      await inAct(async () => { for (let r = 0; r < 8; r += 1) await Promise.resolve(); });
    },
  };
}
const address = { hash: () => "", clearFragment() {}, go() {}, replace() {}, onChange: () => () => undefined };
const FAIL = { ok: false, failure: { kind: "unreachable", reason: "network", message: "Brak połączenia." } };
const ACCOUNT = { ok: true, value: { id: "a1", email: "ala@example.com", emailConfirmed: true } };

test("retry: focus at each refusal, same button, waiting text", async () => {
  await withPage(async ({ root }) => {
    const h = held();
    const { startApp } = await import("../../dist/browser/view/app.js");
    await inAct(() => startApp(document.body, h.client, address));
    await inAct(async () => { for (let r = 0; r < 8; r += 1) await Promise.resolve(); });
    const section = () => document.querySelector(".session-check");
    assert.equal(section().textContent, "Sprawdzanie logowania…");
    assert.equal(section().querySelectorAll("button").length, 0);
    await h.answer("currentAccount", FAIL);
    const button = () => [...section().querySelectorAll("button")];
    assert.equal(button().length, 1);
    assert.equal(button()[0].textContent, "Spróbuj ponownie");
    assert.equal(document.activeElement, button()[0], "focus at the first refusal");
    assert.equal(section().querySelector("p.form-warning").textContent, "Brak połączenia.");
    assert.equal(section().querySelector(".session-check-text"), null);
    const first = button()[0];
    await press(first);
    assert.equal(button().length, 1);
    assert.equal(button()[0], first, "same element while retrying");
    assert.equal(first.textContent, "Sprawdzanie…");
    assert.equal(first.disabled, true);
    assert.equal(section().querySelector("p.form-warning"), null);
    await h.answer("currentAccount", ACCOUNT);
    await h.answer("loadSavedGame", FAIL);
    assert.equal(button()[0], first, "same element again after the second refusal");
    assert.equal(first.textContent, "Spróbuj ponownie");
    assert.equal(first.disabled, false);
    assert.equal(document.activeElement, first, "focus back after a refusal after retrying");
    assert.equal(section().querySelector("p.form-warning").textContent, "Brak połączenia.");
  });
});
