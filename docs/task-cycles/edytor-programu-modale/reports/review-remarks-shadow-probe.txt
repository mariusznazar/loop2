"""Read the darkening the window's shadow lays on the column under its lower edge, from the
author's 'after' screenshot at 1600 x 1000, and the contrast of text roles laid under that shadow."""

import sys

from PIL import Image


def lin(c):
    c = c / 255
    return c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4


def lum(rgb):
    r, g, b = (lin(v) for v in rgb)
    return 0.2126 * r + 0.7152 * g + 0.0722 * b


def ratio(a, b):
    la, lb = lum(a), lum(b)
    hi, lo = max(la, lb), min(la, lb)
    return (hi + 0.05) / (lo + 0.05)


def hexrgb(h):
    return tuple(int(h[i : i + 2], 16) for i in (1, 3, 5))


path, x, y0 = sys.argv[1], int(sys.argv[2]), int(sys.argv[3])
img = Image.open(path).convert("RGB")
panel = hexrgb("#18181b")
roles = {"text": "#d4d4d8", "text-secondary": "#9f9fa9", "text-label": "#8d8d97"}
out = []
for d in range(-3, 45, 3):
    px = img.getpixel((x, y0 + d))
    # Black laid at opacity a over panel: channel * (1 - a); estimate a from the green channel.
    k = px[1] / panel[1] if panel[1] else 1.0
    k = min(k, 1.0)
    cells = []
    for name, h in roles.items():
        t = tuple(round(v * k) for v in hexrgb(h))
        bg = tuple(round(v * k) for v in panel)
        cells.append(f"{name} {ratio(t, bg):.2f}")
    out.append(f"d={d:+3d} px={px} k={k:.2f} " + "; ".join(cells))
with open(sys.argv[4], "w", encoding="utf-8") as f:
    f.write("\n".join(out) + "\n")
print(len(out))
